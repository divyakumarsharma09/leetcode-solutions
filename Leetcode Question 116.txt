class Solution {
    public Node connect(Node root) {
        if (root == null) {
            return null;
        }
         Node levelStart = root;
         while (levelStart.left != null) {
            Node node = levelStart;
            while (node != null) {
                node.left.next = node.right;
                if (node.next != null) {
                    node.right.next = node.next.left;
                }
                node = node.next;
            }
            levelStart = levelStart.left;
        }
        return root;
    }
}
