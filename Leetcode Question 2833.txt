class Solution {
    public int furthestDistanceFromOrigin(String moves) {
        int position = 0;
        int unknown = 0;

        for (char move : moves.toCharArray()) {
            if (move == 'L') {
                position--;
            } else if (move == 'R') {
                position++;
            } else {
                unknown++;
            }
        }

        return Math.abs(position) + unknown;
    }
}
