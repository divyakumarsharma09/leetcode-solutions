class Solution {
    public boolean digitCount(String num) {
        int[] actualCount = new int[10];
        for (char c : num.toCharArray()) {
            actualCount[c - '0']++;
        }
        for (int i = 0; i < num.length(); i++) {
            int claimedCount = num.charAt(i) - '0';
            if (actualCount[i] != claimedCount) {
                return false;
            }
        }
        return true;
    }
}
